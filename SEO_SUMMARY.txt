🎯 USAGE EXAMPLES

Setting Page Metadata:
  import { generateMetadata } from '@/lib/seo'
  
  export const metadata = generateMetadata({
    title: 'Page Title',
    description: 'Page description',
    keywords: ['keyword1', 'keyword2'],
    path: '/page-slug',
  })

Adding JSON-LD Schema:
  import { JsonLdSchema } from '@/components/common/JsonLdSchema'
  import { getWebPageSchema } from '@/lib/seo'
  
  export default function MyPage() {
    return (
      <>
        <JsonLdSchema schema={getWebPageSchema({...})} />
        {/* content */}
      </>
    )
  }

Adding ARIA Labels:
  <section aria-label="Description" role="region">
    <h2>Title</h2>
  </section>

Image Alt Text:
  import { imageAltTexts } from '@/lib/images'
  
  <img src="..." alt={imageAltTexts.hero} />
